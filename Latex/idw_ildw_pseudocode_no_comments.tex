\noindent\textbf{Notation.}
Each link $\ell$ is the segment $[p_{\ell,1},p_{\ell,2}]$ with endpoints
$p_{\ell,1},p_{\ell,2}\in\RR^2$, midpoint
$m_\ell = (p_{\ell,1}+p_{\ell,2})/2$, and length
$L_\ell = \|p_{\ell,2}-p_{\ell,1}\|_2$.
Each pixel $p$ has center $c_p\in\RR^2$.

\section*{IDW}
\begin{algorithm}[h]
\caption{Discretized Midpoint IDW}
\begin{algorithmic}[1]
\Require links $(p_{\ell,1},p_{\ell,2},A_\ell,f_\ell,\mathrm{pol}_\ell)$, radius $r_{\max}$
\For{each link $\ell$}
\State $L_\ell \gets \|p_{\ell,2}-p_{\ell,1}\|_2$
\State $(\kappa_\ell,\alpha_\ell) \gets \mathrm{ITU838}(f_\ell,\mathrm{pol}_\ell)$
\State $R_\ell \gets \left(\dfrac{A_\ell/L_\ell}{\kappa_\ell}\right)^{1/\alpha_\ell}$
\EndFor
\For{each pixel $p$ with center $c_p$}
\State $m_\ell \gets (p_{\ell,1}+p_{\ell,2})/2$ for each link $\ell$
\If{one or more link midpoints lie inside pixel $p$}
\State $\hat R_p \gets$ mean of $R_\ell$ of those links
\Else
\State $N_p \gets \{\ell:\|m_\ell-c_p\|_2\le r_{\max}\}$
\If{$N_p=\emptyset$}
\State $\hat R_p \gets 0$
\Else
\State $\hat R_p \gets \sum_{\ell\in N_p}\dfrac{\|m_\ell-c_p\|_2^{-2}}{\sum_{j\in N_p}\|m_j-c_p\|_2^{-2}} R_\ell$
\EndIf
\EndIf
\EndFor
\State Return field $\hat R$
\end{algorithmic}
\end{algorithm}

\section*{ILDW}
\begin{algorithm}[h]
\caption{Segment-Distance ILDW}
\begin{algorithmic}[1]
\Require links $(p_{\ell,1},p_{\ell,2},A_\ell,f_\ell,\mathrm{pol}_\ell)$, radius $r_{\max}$
\For{each link $\ell$}
\State $L_\ell \gets \|p_{\ell,2}-p_{\ell,1}\|_2$
\State $(\kappa_\ell,\alpha_\ell) \gets \mathrm{ITU838}(f_\ell,\mathrm{pol}_\ell)$
\State $R_\ell \gets \left(\dfrac{A_\ell/L_\ell}{\kappa_\ell}\right)^{1/\alpha_\ell}$
\EndFor
\For{each pixel $p$ with center $c_p$}
\If{one or more links cross pixel $p$}
\State $\hat R_p \gets$ mean of $R_\ell$ of those links
\Else
\State $N_p \gets \{\ell : \mathrm{dist}(c_p,[p_{\ell,1},p_{\ell,2}]) \le r_{\max}\}$
\If{$N_p=\emptyset$}
\State $\hat R_p \gets 0$
\Else
\State $\hat R_p \gets \sum_{\ell\in N_p}\dfrac{\mathrm{dist}(c_p,[p_{\ell,1},p_{\ell,2}])^{-2}}{\sum_{j\in N_p}\mathrm{dist}(c_p,[p_{j,1},p_{j,2}])^{-2}} R_\ell$
\EndIf
\EndIf
\EndFor
\State Return field $\hat R$
\end{algorithmic}
\end{algorithm}
