\section*{IDW (midpoint-based, truncated)}
\begin{algorithm}[h]
\caption{IDW\_Field\_From\_Est\_Input}
\begin{algorithmic}[1]
\Require \texttt{est\_input\_json}, $r_{\max}$, power $p$, $\varepsilon$, default value $d_0$, optional \texttt{link\_values}
\State Read \texttt{header}, \texttt{links} from \texttt{est\_input\_json}
\State Compute midpoint of each link: $m_k = \big((x0_k+x1_k)/2,\,(y0_k+y1_k)/2\big)$
\If{\texttt{link\_values} is not provided}
\State $A_k \gets$ link attenuation \texttt{A\_db}, $L_k \gets$ link length in km
\State $(k_k,\alpha_k) \gets \texttt{itu838\_k\_alpha}(f_k,\text{pol}_k)$
\State $R_k \gets \texttt{link\_rain\_from\_attenuation}(A_k,L_k,k_k,\alpha_k)$
\Else
\State $R_k \gets$ provided \texttt{link\_values}
\EndIf
\State Build query points $q_t$ as all pixel centers from \texttt{header}
\State Build KD-tree on $\{m_k\}$
\For{each pixel center $q_t$}
\State Initialize $\hat R_t \gets d_0$
\State $I_t \gets \{k:\|m_k-q_t\|\le r_{\max}\}$ via radius query
\If{$I_t$ is empty}
\State \textbf{continue}
\EndIf
\State Compute distances $\delta_k=\|m_k-q_t\|$ for $k\in I_t$
\If{any $\delta_k \le 0$}
\State $\hat R_t \gets R_k$ of first exact hit; \textbf{continue}
\EndIf
\State $\delta_k \gets \max(\delta_k,\varepsilon)$
\State $w_k \gets \delta_k^{-p}$
\If{$\sum_{k\in I_t} w_k \le 0$ or not finite}
\State \textbf{continue}
\EndIf
\State Normalize $w_k \gets w_k/\sum_j w_j$
\State $\hat R_t \gets \sum_{k\in I_t} w_k R_k$
\EndFor
\State Reshape $\hat R$ to $(H,W)$ and return $(\hat R, R)$
\end{algorithmic}
\end{algorithm}

\section*{ILDW (segment-distance + crossing rule)}
\begin{algorithm}[h]
\caption{ILDW\_Field\_From\_Est\_Input}
\begin{algorithmic}[1]
\Require \texttt{est\_input\_json}, $r_{\max}$, power $p$, $\varepsilon$, default value $d_0$, optional \texttt{link\_values}
\State Read \texttt{header}, \texttt{links}, \texttt{segments\_by\_link} from \texttt{est\_input\_json}
\State Compute link endpoints $(a_k,b_k)$ and midpoints $m_k$
\If{\texttt{link\_values} is not provided}
\State Compute $R_k$ from $(A_k,L_k,f_k,\text{pol}_k)$ exactly as in IDW
\Else
\State $R_k \gets$ provided \texttt{link\_values}
\EndIf
\State Build list $C_t$: crossing links per pixel from \texttt{segments\_by\_link}
\State $r_{\text{search}} \gets r_{\max} + \max_k \|b_k-a_k\|/2$
\State Build KD-tree on link midpoints $\{m_k\}$
\State Build all pixel centers $q_t$; initialize $\hat R_t \gets d_0$
\For{each pixel index $t$}
\If{$C_t \neq \emptyset$}
\State Deduplicate $C_t$ (if needed), then set $\hat R_t \gets \text{mean}\{R_k:k\in C_t\}$
\State \textbf{continue}
\EndIf
\State Candidate links $I_t \gets \{k:\|m_k-q_t\|\le r_{\text{search}}\}$ via midpoint radius query
\If{$I_t$ is empty}
\State \textbf{continue}
\EndIf
\State Compute exact point-to-segment distances $\delta_k=\text{dist}(q_t,[a_k,b_k])$ for $k\in I_t$
\State Keep only $J_t=\{k\in I_t:\delta_k\le r_{\max}\}$
\If{$J_t$ is empty}
\State \textbf{continue}
\EndIf
\If{any $\delta_k \le 0$ in $J_t$}
\State $\hat R_t \gets R_k$ of first exact hit; \textbf{continue}
\EndIf
\State $\delta_k \gets \max(\delta_k,\varepsilon)$ for $k\in J_t$
\State $w_k \gets \delta_k^{-p}$; if $\sum w_k$ invalid then \textbf{continue}
\State Normalize and set $\hat R_t \gets \sum_{k\in J_t} w_k R_k$
\EndFor
\State Reshape $\hat R$ to $(H,W)$ and return $(\hat R, R)$
\end{algorithmic}
\end{algorithm}
