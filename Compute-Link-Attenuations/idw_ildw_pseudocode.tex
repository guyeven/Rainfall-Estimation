\noindent\textbf{Coordinate convention (used in both algorithms).}
All points are in local patch coordinates, in meters, with origin at the NW (top-left) corner.
$x$ increases to the east, $y$ increases to the south.
Thus $(x0_k,y0_k)$ and $(x1_k,y1_k)$ are link endpoints in this frame, and $q_t$ is the center of pixel $t$ in the same frame.

\section*{IDW}
\begin{algorithm}[h]
\caption{Discretized Midpoint IDW}
\begin{algorithmic}[1]
\Require links $(x0_k,y0_k,x1_k,y1_k,A_k,f_k,\text{pol}_k)$, radius $r_{\max}$ (default: $15000$ m), power $p$ (default: $2$), default $d_0$ (default: $0$)
\For{each link $k$}
\State midpoint $m_k \gets \big((x0_k+x1_k)/2,\,(y0_k+y1_k)/2\big)$
\State $\ell_k^{(m)} \gets \| (x1_k,y1_k)-(x0_k,y0_k) \|$ \Comment{computed in meters}
\State $L_k \gets \ell_k^{(m)}/1000$ \Comment{converted and stored in km}
\State $(\kappa_k,\alpha_k) \gets \mathrm{ITU838}(f_k,\text{pol}_k)$
\State $R_k \gets \left(\dfrac{A_k/L_k}{\kappa_k}\right)^{1/\alpha_k}$ \Comment{link value, located at midpoint $m_k$}
\EndFor
\For{each pixel $t$ with center $q_t$}
\If{one or more link midpoints lie inside pixel $t$}
\State $\hat R_t \gets$ mean of $R_k$ of those links
\Else
\State $N_t \gets \{k:\|m_k-q_t\|\le r_{\max}\}$
\If{$N_t=\emptyset$}
\State $\hat R_t \gets d_0$
\Else
\State $\tilde w_k \gets \|m_k-q_t\|^{-p}$ for $k\in N_t$
\State $w_k \gets \dfrac{\tilde w_k}{\sum_{j\in N_t}\tilde w_j}$ for $k\in N_t$
\State $\hat R_t \gets \sum_{k\in N_t} w_k R_k$
\EndIf
\EndIf
\EndFor
\State Return field $\hat R$
\end{algorithmic}
\end{algorithm}

\section*{ILDW}
\begin{algorithm}[h]
\caption{Segment-Distance ILDW}
\begin{algorithmic}[1]
\Require links $(a_k,b_k,A_k,f_k,\text{pol}_k)$, radius $r_{\max}$ (default: $15000$ m), power $p$ (default: $2$), default $d_0$ (default: $0$)
\For{each link $k$}
\State midpoint $m_k \gets (a_k+b_k)/2$
\State $\ell_k^{(m)} \gets \|b_k-a_k\|$ \Comment{computed in meters}
\State $L_k \gets \ell_k^{(m)}/1000$ \Comment{converted and stored in km}
\State $(\kappa_k,\alpha_k) \gets \mathrm{ITU838}(f_k,\text{pol}_k)$
\State $R_k \gets \left(\dfrac{A_k/L_k}{\kappa_k}\right)^{1/\alpha_k}$ \Comment{link value; used on segment $[a_k,b_k]$}
\EndFor
\For{each pixel $t$ with center $q_t$}
\If{one or more links cross pixel $t$}
\State $\hat R_t \gets$ mean of $R_k$ of those links
\Else
\State $N_t \gets \{k : \mathrm{dist}(q_t,[a_k,b_k]) \le r_{\max}\}$ \Comment{$\mathrm{dist}$ is point-to-segment distance}
\If{$N_t=\emptyset$}
\State $\hat R_t \gets d_0$
\Else
\State $\tilde w_k \gets \mathrm{dist}(q_t,[a_k,b_k])^{-p}$ for $k\in N_t$
\State $w_k \gets \dfrac{\tilde w_k}{\sum_{j\in N_t}\tilde w_j}$ for $k\in N_t$
\State $\hat R_t \gets \sum_{k\in N_t} w_k R_k$
\EndIf
\EndIf
\EndFor
\State Return field $\hat R$
\end{algorithmic}
\end{algorithm}
