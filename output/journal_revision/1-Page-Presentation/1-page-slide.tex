      %\textbf{Story (5G backhaul).}\\
      \textbf{Opportunity:} Rain attenuates 5G Microwave backhaul links
      \\
      \vspace{0.25em}
      \textbf{Forward Model (Rain $\Rightarrow$ Attenuation).}
      For link $\ell$
      \[
        A_\ell = \int_{\ell} k(r(x))\,ds,
        \qquad k(r)=\alpha r^\beta.
      \]

      \vspace{0.25em}
      \textbf{Question (Inverse Problem).}
      Recover the rain field?
      \\
      \vspace{0.35em}
      \textbf{Method.}
      \begin{itemize}
        \item Create Real Benchmark: \\200 rainfall scenes on $\sim 60\times 60$ km rectangles
        \item Overlay real NL microwave links on each scene
        \item Compute attenuations $\{A_\ell\}$ (forward model)
        \item Inverse problem: Estimate rain field.
      \end{itemize}
